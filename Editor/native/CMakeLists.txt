cmake_minimum_required(VERSION 3.20)
project(UIEditorLuau LANGUAGES CXX C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MINGW)
  add_compile_options(-Wno-attributes)
endif()
include(FetchContent)
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_STATIC_CRT ON CACHE BOOL "" FORCE)
FetchContent_Declare(luau GIT_REPOSITORY https://github.com/luau-lang/luau.git GIT_TAG bb3956547affc1af7b7d196f96756895255dd05d)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
FetchContent_MakeAvailable(luau json)
add_executable(ui-luau main.cpp)
target_compile_features(ui-luau PRIVATE cxx_std_17)
target_link_libraries(ui-luau PRIVATE Luau.Compiler Luau.VM nlohmann_json::nlohmann_json)
if(MINGW)
  target_link_options(ui-luau PRIVATE -static)
endif()
